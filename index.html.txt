                </p>

                    EQUATION 04
                </span>

                <h3>
                    Solid Disc
                </h3>

                </div>

                <p>

                </p>

                <span>
                    EQUATION 05
                </span>

                <h3>
                    Thin Ring

                </div>

                <p>

                <span>
                    FINAL MODEL
                </span>
